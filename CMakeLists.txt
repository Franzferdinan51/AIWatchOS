# AIWatchOS — an AI-first watch OS for the Waveshare ESP32-S3-Touch-AMOLED-2.06
# (ESP32-S3R8, 410x502 CO5300 AMOLED, FT3168 touch).
cmake_minimum_required(VERSION 3.16)

set(EXTRA_COMPONENT_DIRS $ENV{IDF_PATH}/examples/common_components/protocol_examples_common components/aiwatchos_os components/aiwatchos_net components/aiwatchos_weather components/noise_core components/minimp3 apps/muse apps/hermes)

# Pull in the ESP-IDF build system. Without this include, `idf.py build` only
# runs a bare CMake configure and never compiles any firmware (no IDF targets,
# no components, no project_description.json).
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(aiwatchos)

# Bake the UI assets (baked Inter fonts + shadow sprites from
# tools/gen_font.py, committed under assets/) into the aiwatchos_assets
# FAT partition at build time. FLASH_IN_PROJECT makes the partition
# image part of `idf.py flash`, so the fonts land with the firmware
# while the app slots stay untouched. The OS mounts the partition
# read-only at boot (font.cpp) and loads the assets into PSRAM.
# The font .bin files are BUILD INPUTS generated by tools/gen_font.py.
# They were silently absent from every fresh clone for a long time
# (the *.bin rule in .gitignore kept them out of git while comments
# here claimed they were "committed under assets/"), so a fresh-clone
# flash produced a valid-but-EMPTY assets image and a watch with no
# fonts (first boot 2026-10-13). Fail the configure loudly instead of
# ever baking an empty image again.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/font_body18.bin")
    message(FATAL_ERROR
        "assets/font_*.bin missing — run `python3 tools/gen_font.py` "
        "in the repo root before building (generates the baked Inter "
        "fonts + sprites into assets/).")
endif()
# Plain (non-WL) image: the runtime mounts it RAW via
# esp_vfs_fat_spiflash_mount_ro, so the FAT boot sector must sit at
# partition offset 0. fatfs_create_spiflash_image wraps the volume in
# a wear-levelling layer (boot sector at +0x1000) and the raw mount
# then fails with ESP_FAIL — that mismatch shipped silently because
# assets_load was compiling its host branch (see font.cpp).
fatfs_create_partition_image(aiwatchos_assets assets FLASH_IN_PROJECT)
